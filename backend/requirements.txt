# Direct dependencies only.
#
# This file previously contained `pip freeze` output, which pinned transitive
# packages (cffi, pycparser, h11, wsproto, msgspec, greenlet, urllib3, ...) to
# versions that have no wheels for current Python releases. That made a clean
# install fail with a compiler error on any machine without MSVC / build
# headers, even though nothing in this project imports those packages directly.
#
# Pin what we actually import; let pip resolve the rest. The Docker image
# pins the Python version (3.11), which is what keeps production builds
# reproducible.

# Core framework

flask-cors==5.0.1

Werkzeug==3.1.8

# Database
SQLAlchemy==2.0.52
# 2.9.12 is the first release with cp314 wheels; earlier pins force a
# source build that needs libpq headers and a C compiler.

# Async / WebSocket support

# required for Flask-SocketIO's WebSocket transport. 0.41.1 is the first
# release declaring Python 3.14 support.

# HTTP client (used for the exchange-rate lookup in app/extension.py)

# Environment & config
